function getLocalStorageJSON(key,fallback){

try{

const data =
JSON.parse(
localStorage.getItem(key) || ""
);

return data ?? fallback;

}catch{

return fallback;

}

}


/* ==========================================
   ПЕРЕВІРКА АДМІНА
========================================== */

async function checkAdmin(){

const {
data:{
session
}
} =
await supabase.auth.getSession();


if(!session){

window.location.href="index.html";

return false;

}


const {
data:profile,
error
} =
await supabase
.from("profiles")
.select("role")
.eq("id",session.user.id)
.single();


if(error || !profile || profile.role!=="admin"){

window.location.href="index.html";

return false;

}


return true;

}

/* ==========================================
   СИНХРОНІЗАЦІЯ ЖАНРІВ
========================================== */

async function migrateGenres(){

log("🏷️ Перевіряю жанри...");


const localGenres =
getLocalGenres();


const {
data:existing,
error
} =
await supabase
.from("genres")
.select("id,code,name");


if(error){

throw new Error(
"Не вдалося отримати жанри: " +
error.message
);

}


const genreMap={};


existing.forEach(genre=>{

genreMap[genre.code]=genre.id;

});


for(const genre of localGenres){

if(genreMap[genre.id]){

continue;

}


const {
data,
error:insertError
} =
await supabase
.from("genres")
.insert({

name:genre.name,

code:genre.id

})
.select("id,code")
.single();


if(insertError){

throw new Error(
`Не вдалося створити жанр "${genre.name}": ${insertError.message}`
);

}


genreMap[data.code]=data.id;

log(
`✅ Додано жанр: ${genre.emoji || "📚"} ${genre.name}`
);

}


return genreMap;

}


/* ==========================================
   ОТРИМАННЯ КНИГ
========================================== */

function getAllLocalBooks(){

const edits =
getLocalStorageJSON(
"bookEdits",
{}
);


const userBooks =
getLocalStorageJSON(
"myBooks",
[]
);


const deleted =
getLocalStorageJSON(
"deletedStandardBooks",
[]
);


const books=[];


/*
    Стандартні книги.
    Навіть видалені додаємо до Supabase,
    але ставимо is_deleted=true.
*/

standardBooks.forEach(book=>{

const finalBook={

...book,

...(edits[book.id] || {})

};


books.push({

...finalBook,

id:book.id,

is_deleted:
deleted.includes(book.id)

});

});


/*
    Власні книги користувача
*/

userBooks.forEach(book=>{

books.push({

...book,

is_deleted:false

});

});


return books;

}


/* ==========================================
   ПІДГОТОВКА КНИГИ
========================================== */

function prepareBook(book,genreMap){

const genreId =
genreMap[book.genre] || null;


return {

book_code:String(book.id),

title:String(book.title || ""),

author:String(book.author || ""),

genre_id:genreId,

description:String(book.description || ""),

image_url:String(
book.image ||
book.image_url ||
""
),

price:String(book.price || ""),

year:String(book.year || ""),

pages:String(book.pages || ""),

language:String(book.language || ""),

publisher:String(book.publisher || ""),

isbn:String(book.isbn || ""),

rating:String(book.rating || ""),

series:String(book.series || ""),

buy_link:String(
book.buyLink ||
book.buy_link ||
""
),

is_deleted:Boolean(
book.is_deleted
),

updated_at:new Date().toISOString()

};

}


/* ==========================================
   МІГРАЦІЯ КНИГ
========================================== */

async function migrateBooks(genreMap){

const books =
getAllLocalBooks();


log(
`📚 Знайдено книг для перенесення: ${books.length}`
);


let success=0;


for(const book of books){

const prepared =
prepareBook(
book,
genreMap
);


const {
error
} =
await supabase
.from("books")
.upsert(
prepared,
{
onConflict:"book_code"
}
);


if(error){

throw new Error(
`Книга "${book.title}" не перенесена: ${error.message}`
);

}


success++;


log(
`✅ ${success}/${books.length} — ${book.title}`
);

}


return success;

}

async function startMigration(){

const button =
document.getElementById(
"migrate-button"
);


button.disabled=true;

document.getElementById(
"progress"
).textContent =
"Починаю міграцію...";


try{

const isAdmin =
await checkAdmin();


if(!isAdmin){

return;

}


log("🔐 Адміністратора підтверджено.");


log("🏷️ Синхронізація жанрів...");


const genreMap =
await migrateGenres();


log(
`✅ Жанрів готово: ${Object.keys(genreMap).length}`
);


log("📚 Починаю перенесення книг...");


const count =
await migrateBooks(genreMap);


log("");


log(
`🎉 ГОТОВО! Перенесено книг: ${count}`
);


log(
"Тепер книги зберігаються у Supabase."
);


document.getElementById(
"progress"
).classList.add("success");


button.textContent =
"✅ Міграцію завершено";


}catch(error){

console.error(error);


log("");


log(
"❌ ПОМИЛКА:"
);


log(error.message);


document.getElementById(
"progress"
).classList.add("error");


button.disabled=false;

button.textContent =
"🔄 Спробувати ще раз";

}

}
